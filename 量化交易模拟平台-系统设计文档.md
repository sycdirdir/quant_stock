| `kline:get` | Renderer → Main | 获取K线数据 |
| `data:download` | Renderer → Main | 下载数据到本地 |

### 2.3 本地 SQLite 数据库设计

客户端本地使用 SQLite，存储下载的K线数据和用户策略。

```sql
-- 股票基础信息（缓存，用于搜索/展示）
CREATE TABLE stocks (
    ts_code    VARCHAR(20) PRIMARY KEY,  -- 如 000001.SZ
    symbol     VARCHAR(10),
    name       VARCHAR(100),
    area       VARCHAR(50),
    industry   VARCHAR(100),
    market     VARCHAR(20),
    list_date  VARCHAR(10),
    is_hs      VARCHAR(1)                -- H、S、 N
);

-- 日线数据（从服务端下载）
CREATE TABLE daily (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    ts_code    VARCHAR(20) NOT NULL,
    trade_date VARCHAR(10) NOT NULL,     -- YYYYMMDD
    open       DECIMAL(10,3),
    high       DECIMAL(10,3),
    low        DECIMAL(10,3),
    close      DECIMAL(10,3),
    pre_close  DECIMAL(10,3),
    change     DECIMAL(10,3),
    pct_chg    DECIMAL(10,4),
    vol        DECIMAL(20,2),
    amount     DECIMAL(20,2),
    UNIQUE(ts_code, trade_date)
);
CREATE INDEX idx_daily_ts_code ON daily(ts_code);
CREATE INDEX idx_daily_trade_date ON daily(trade_date);
CREATE INDEX idx_daily_ts_date ON daily(ts_code, trade_date);

-- 周线数据
CREATE TABLE weekly (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    ts_code    VARCHAR(20) NOT NULL,
    trade_date VARCHAR(10) NOT NULL,
    open       DECIMAL(10,3),
    high       DECIMAL(10,3),
    low        DECIMAL(10,3),
    close      DECIMAL(10,3),
    pre_close  DECIMAL(10,3),
    change     DECIMAL(10,3),
    pct_chg    DECIMAL(10,4),
    vol        DECIMAL(20,2),
    amount     DECIMAL(20,2),
    UNIQUE(ts_code, trade_date)
);
CREATE INDEX idx_weekly_ts_code ON weekly(ts_code);
CREATE INDEX idx_weekly_trade_date ON weekly(trade_date);

-- 月线数据
CREATE TABLE monthly (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    ts_code    VARCHAR(20) NOT NULL,
    trade_date VARCHAR(10) NOT NULL,
    open       DECIMAL(10,3),
    high       DECIMAL(10,3),
    low        DECIMAL(10,3),
    close      DECIMAL(10,3),
    pre_close  DECIMAL(10,3),
    change     DECIMAL(10,3),
    pct_chg    DECIMAL(10,4),
    vol        DECIMAL(20,2),
    amount     DECIMAL(20,2),
    UNIQUE(ts_code, trade_date)
);
CREATE INDEX idx_monthly_ts_code ON monthly(ts_code);
CREATE INDEX idx_monthly_trade_date ON monthly(trade_date);

-- 策略模板（JSON存储配置 + Python代码）
CREATE TABLE strategies (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id      VARCHAR(64),                        -- 微信openid
    name         VARCHAR(100) NOT NULL,
    description  TEXT,
    config_json  TEXT NOT NULL,                      -- 可视化配置
    code         TEXT,                               -- Python代码（可选）
    created_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
    synced       INTEGER DEFAULT 0                   -- 0=未同步 1=已同步
);
CREATE INDEX idx_strategies_user ON strategies(user_id);

-- 回测结果
CREATE TABLE backtest_results (
    id               INTEGER PRIMARY KEY AUTOINCREMENT,
    strategy_id      INTEGER NOT NULL,
    strategy_name    VARCHAR(100),
    ts_code          VARCHAR(20) NOT NULL,
    stock_name       VARCHAR(100),
    period           VARCHAR(10) NOT NULL,            -- daily/weekly/monthly
    start_date       VARCHAR(10),
    end_date         VARCHAR(10),
    initial_cash     DECIMAL(15,2) DEFAULT 100000,
    result_json      TEXT NOT NULL,                  -- 完整结果（包含 trades, equity_curve）
    summary_json     TEXT NOT NULL,                  -- 摘要（核心指标）
    created_at       DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (strategy_id) REFERENCES strategies(id)
);
CREATE INDEX idx_backtest_strategy ON backtest_results(strategy_id);
CREATE INDEX idx_backtest_ts_code ON backtest_results(ts_code);

-- 同步记录
CREATE TABLE sync_log (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    sync_type    VARCHAR(20),                        -- data/strategy
    last_sync    DATETIME,
    synced_items TEXT,                               -- JSON数组
    version      VARCHAR(20)
);

-- 认证信息
CREATE TABLE auth (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id      VARCHAR(64) PRIMARY KEY,            -- openid
    nickname     VARCHAR(100),
    avatar_url   VARCHAR(500),
    jwt_token    TEXT,
    token_expire DATETIME
);
```

### 2.4 回测引擎设计（Python 子进程）

回测计算在 Node.js 主进程中通过 `child_process.spawn` 启动 Python 子进程执行，保证性能。

```
Main Process (Node.js)
      │
      │  stdin: JSON命令
      │  stdout: JSON结果
      ▼
Python 子进程 (backtest_engine.py)
      │
      │  读取 SQLite 中的 K 线数据
      │
      ▼
┌─────────────────────────────────────────┐
│             回测核心逻辑                   │
│                                         │
│  1. 加载数据 (pandas.read_sqlite)        │
│  2. 计算指标 (ta-lib: MA, MACD, KDJ)    │
│  3. 信号计算 (均线交叉等条件判断)         │
│  4. 模拟交易 (买入/卖出/持仓)             │
│  5. 风控检查 (止损/止盈/仓位)             │
│  6. 指标计算 (夏普/最大回撤/胜率)         │
│                                         │
└─────────────────────────────────────────┘
      │
      │  stdout: JSON
      ▼
主进程解析结果 → IPC → Renderer 更新UI
```

**Python 子进程接口（stdin/stdout JSON 协议）：**


```sql
-- 使用现有 tushare 库，新建 app schema
CREATE SCHEMA IF NOT EXISTS app;

    id           SERIAL PRIMARY KEY,

    config_json  TEXT NOT NULL,
    code         TEXT,
    created_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, strategy_id)
);
CREATE INDEX idx_cloud_strategies_user ON app.cloud_strategies(user_id);

CREATE TABLE app.data_update_records (
    id           SERIAL PRIMARY KEY,
```

### 5.3 交易模拟模块

```python
# backtest_engine/trading.py

from dataclasses import dataclass
from typing import Optional

@dataclass
class Trade:
    buy_date: str
    buy_price: float
    sell_date: Optional[str]
    sell_price: Optional[float]
    profit: Optional[float]
    profit_pct: Optional[float]
    holding_days: Optional[int]

class Simulator:
    def __init__(self, initial_cash: float, risk_config: dict):
        self.cash = initial_cash
        self.initial_cash = initial_cash
        self.position = 0           # 持仓股数
        self.position_price = 0     # 持仓成本
        self.position_date = None   # 持仓日期
        self.risk = risk_config
        self.trades = []
        self.equity_curve = []
        self.daily_trades = {}      # 当日交易次数统计

    def buy(self, date: str, price: float):
        """买入"""
        # 风控检查
        if self.position > 0:
            return  # 已有持仓，不买入

        position_size = self.risk['position_size']
        shares = int(position_size / price)

        if shares * price > self.cash:
            shares = int(self.cash / price)

        if shares <= 0:
            return

        self.cash -= shares * price
        self.position = shares
        self.position_price = price
        self.position_date = date

        # 记录当日交易
        self.daily_trades[date] = self.daily_trades.get(date, 0) + 1

    def sell(self, date: str, price: float, reason: str = ""):
        """卖出"""
        if self.position == 0:
            return

        sell_value = self.position * price
        profit = sell_value - (self.position * self.position_price)
        profit_pct = profit / (self.position * self.position_price)

        self.trades.append({
            'buy_date': self.position_date,
            'buy_price': self.position_price,
            'sell_date': date,
            'sell_price': price,
            'profit': profit,
            'profit_pct': profit_pct,
            'holding_days': (pd.to_datetime(date) - pd.to_datetime(self.position_date)).days,
            'sell_reason': reason
        })

        self.cash += sell_value
        self.position = 0
        self.position_price = 0
        self.position_date = None

        # 记录当日交易
        self.daily_trades[date] = self.daily_trades.get(date, 0) + 1

    def check_risk(self, date: str, price: float) -> bool:
        """风控检查，返回是否触发风控"""
        if self.position == 0:
            return False

        # 止损
        loss_pct = (self.position_price - price) / self.position_price
        if loss_pct >= self.risk['stop_loss']:
            self.sell(date, price, reason=f"止损-{loss_pct:.2%}")
            return True

        # 止盈
        profit_pct = (price - self.position_price) / self.position_price
        if profit_pct >= self.risk['take_profit']:
            self.sell(date, price, reason=f"止盈-{profit_pct:.2%}")
            return True

        # 单日最大交易次数
        if self.daily_trades.get(date, 0) >= self.risk['max_daily_trades']:
            return False  # 达到上限，不交易

        # 最大持仓比例
        total_value = self.cash + self.position * price
        position_ratio = (self.position * price) / total_value
        if position_ratio > self.risk['max_position_ratio']:
            return False  # 超过仓位上限

        return False

    def get_equity(self, date: str, close_price: float) -> float:
        """获取当前权益"""
        return self.cash + self.position * close_price

    def finalize(self):
        """平仓并计算结果"""
        # 如果还有持仓，按最后一天收盘价平仓
        if self.position > 0:
            last_trade = self.trades[-1] if self.trades else None
            if last_trade:
                self.sell(last_trade['sell_date'], last_trade['sell_price'], reason="回测结束")
```

### 5.4 指标计算模块

```python
# backtest_engine/metrics.py

import pandas as pd
import numpy as np

def calculate_metrics(simulator: Simulator, initial_cash: float) -> dict:
    """计算回测核心指标"""
    trades = simulator.trades
    equity_curve = simulator.equity_curve

    if not trades:
        return {
            "total_return": 0,
            "annual_return": 0,
            "sharpe_ratio": 0,
            "max_drawdown": 0,
            "win_rate": 0,
            "profit_loss_ratio": 0,
            "total_trades": 0,
            "avg_holding_days": 0
        }

    # 基本收益
    final_equity = simulator.cash
    total_return = (final_equity - initial_cash) / initial_cash

    # 年化收益率
    first_trade_date = pd.to_datetime(trades[0]['buy_date'])
    last_trade_date = pd.to_datetime(trades[-1]['sell_date'])
    years = (last_trade_date - first_trade_date).days / 365
    annual_return = total_return / years if years > 0 else 0

    # 夏普比率
    if len(equity_curve) > 1:
        returns = pd.Series([e['return'] for e in equity_curve if 'return' in e])
        if returns.std() > 0:
            sharpe_ratio = (returns.mean() - 0.03) / returns.std() * np.sqrt(252)  # 假设无风险利率3%
        else:
            sharpe_ratio = 0
    else:
        sharpe_ratio = 0

    # 最大回撤
    equity_series = pd.Series([e['value'] for e in equity_curve])
    cummax = equity_series.cummax()
    drawdown = (equity_series - cummax) / cummax
    max_drawdown = drawdown.min()

    # 胜率
    winning_trades = [t for t in trades if t.get('profit', 0) > 0]
    win_rate = len(winning_trades) / len(trades) if trades else 0

    # 盈亏比
    avg_profit = np.mean([t['profit'] for t in winning_trades]) if winning_trades else 0
    losing_trades = [t for t in trades if t.get('profit', 0) <= 0]
    avg_loss = abs(np.mean([t['profit'] for t in losing_trades])) if losing_trades else 1
    profit_loss_ratio = avg_profit / avg_loss if avg_loss > 0 else 0

    # 平均持仓时长
    holding_days = [t['holding_days'] for t in trades if t.get('holding_days')]
    avg_holding_days = np.mean(holding_days) if holding_days else 0

    return {
        "total_return": round(total_return, 4),
        "annual_return": round(annual_return, 4),
        "sharpe_ratio": round(sharpe_ratio, 2),
        "max_drawdown": round(max_drawdown, 4),
        "win_rate": round(win_rate, 4),
        "profit_loss_ratio": round(profit_loss_ratio, 2),
        "total_trades": len(trades),
        "avg_holding_days": round(avg_holding_days, 1)
    }
```

```tsx
import React, { useEffect, useRef } from 'react';
```

#### 6.2.2 策略编辑器组件

```tsx
// src/renderer/components/StrategyEditor/index.tsx

import React, { useState } from 'react';
import { useStrategyStore } from '../../store';

interface Condition {
  type: 'ma_cross' | 'kdj_cross' | 'macd_cross' | 'price_break';
  params: Record<string, any>;
  logic: 'AND' | 'OR';
}

export const StrategyEditor: React.FC = () => {
  const [name, setName] = useState('');
  const [conditions, setConditions] = useState<Condition[]>([]);
  const [riskConfig, setRiskConfig] = useState({
    stopLoss: 5,
    takeProfit: 10,
    positionSize: 20000,
    maxDailyTrades: 3,
    maxPositionRatio: 30
  });
  const { saveStrategy } = useStrategyStore();

  const addCondition = (type: Condition['type']) => {
    const defaultParams = {
      ma_cross: { fastMa: 5, slowMa: 20 },
      kdj_cross: { kThreshold: 20, dThreshold: 80 },
      macd_cross: {},
      price_break: { period: 20, direction: 'up' }
    };

    setConditions([
      ...conditions,
      { type, params: defaultParams[type], logic: 'AND' }
    ]);
  };

  const handleSave = async () => {
    const strategy = {
      name,
      conditions,
      riskConfig
    };

    await saveStrategy(strategy);
  };

  return (
    <div className="strategy-editor">
      <div className="strategy-name">
        <input
          value={name}
          onChange={e => setName(e.target.value)}
          placeholder="策略名称"
        />
      </div>

      <div className="conditions">
        <h3>信号条件</h3>
        {conditions.map((c, i) => (
          <div key={i} className="condition-row">
            {i > 0 && (
              <select
                value={c.logic}
                onChange={e => {
                  const updated = [...conditions];
                  updated[i].logic = e.target.value as 'AND' | 'OR';
                  setConditions(updated);
                }}
              >
                <option value="AND">且</option>
                <option value="OR">或</option>
              </select>
            )}
            <span>{c.type}: {JSON.stringify(c.params)}</span>
            <button onClick={() => setConditions(conditions.filter((_, idx) => idx !== i))}>
              删除
            </button>
          </div>
        ))}
        <div className="add-condition">
          <button onClick={() => addCondition('ma_cross')}>+ 均线交叉</button>
          <button onClick={() => addCondition('kdj_cross')}>+ KDJ交叉</button>
          <button onClick={() => addCondition('macd_cross')}>+ MACD交叉</button>
        </div>
      </div>

      <div className="risk-config">
        <h3>风控配置</h3>
        <label>止损 %: <input type="number" value={riskConfig.stopLoss} onChange={...} /></label>
        <label>止盈 %: <input type="number" value={riskConfig.takeProfit} onChange={...} /></label>
        <label>每次买入金额: <input type="number" value={riskConfig.positionSize} onChange={...} /></label>
      </div>

      <button className="save-btn" onClick={handleSave}>保存策略</button>
    </div>
  );
};
```

```tsx
// src/renderer/components/BacktestPanel/index.tsx

import React, { useState } from 'react';
import { useBacktestStore } from '../../store';

interface BacktestPanelProps {
  tsCode: string;
  strategyId: number;
}

export const BacktestPanel: React.FC<BacktestPanelProps> = ({ tsCode, strategyId }) => {
  const [startDate, setStartDate] = useState('20250101');
  const [endDate, setEndDate] = useState('20260424');
  const [running, setRunning] = useState(false);
  const { result, runBacktest, progress } = useBacktestStore();

    setRunning(true);
    await runBacktest({
      tsCode,
      strategyId,
      period: 'daily',
      startDate,
      endDate,
      initialCash: 100000
    });
    setRunning(false);
  };

      <div className="params">

          开始日期:
          <input type="date" value={startDate} onChange={e => setStartDate(e.target.value)} />

          <input type="date" value={endDate} onChange={e => setEndDate(e.target.value)} />
        </label>
```

### 6.3 Zustand Store

```typescript
// src/renderer/store/index.ts

import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface KlineItem {
  trade_date: string;
  open: number;
  high: number;
  low: number;
  close: number;
  vol: number;
  ma5?: number;
  ma10?: number;
  ma20?: number;
}

interface Strategy {
  id: number;
  name: string;
  config_json: string;
  code?: string;
}

interface BacktestResult {
  summary: {
    total_return: number;
    annual_return: number;
    sharpe_ratio: number;
    max_drawdown: number;
    win_rate: number;
    profit_loss_ratio: number;
    total_trades: number;
    avg_holding_days: number;
  };
  equity_curve: { date: string; value: number }[];
  trades: any[];
  signals: any[];
}

interface AppState {
  // 股票
  stocks: any[];
  selectedStock: string | null;

  // K线
  klineData: KlineItem[];
  klineLoading: boolean;

  // 策略
  strategies: Strategy[];

  // 回测
  backtestResult: BacktestResult | null;
  backtestProgress: number;
  backtestRunning: boolean;

  // Auth
  user: any | null;
  jwtToken: string | null;

  // Actions
  setStocks: (stocks: any[]) => void;
  selectStock: (tsCode: string) => void;
  setKlineData: (data: KlineItem[]) => void;
  setKlineLoading: (loading: boolean) => void;
  setStrategies: (strategies: Strategy[]) => void;
  setBacktestResult: (result: BacktestResult | null) => void;
  setBacktestProgress: (progress: number) => void;
  setUser: (user: any) => void;
  setJwtToken: (token: string) => void;
}

export const useAppStore = create<AppState>()(
  persist(
    (set) => ({
      stocks: [],
      selectedStock: null,
      klineData: [],
      klineLoading: false,
      strategies: [],
      backtestResult: null,
      backtestProgress: 0,
      backtestRunning: false,
      user: null,
      jwtToken: null,

      setStocks: (stocks) => set({ stocks }),
      selectStock: (tsCode) => set({ selectedStock: tsCode }),
      setKlineData: (data) => set({ klineData: data }),
      setKlineLoading: (loading) => set({ klineLoading: loading }),
      setStrategies: (strategies) => set({ strategies }),
      setBacktestResult: (result) => set({ backtestResult: result }),
      setBacktestProgress: (progress) => set({ backtestProgress: progress }),
      setUser: (user) => set({ user }),
      setJwtToken: (token) => set({ jwtToken: token }),
    }),
    {
      name: 'quant-app-storage',
      partialize: (state) => ({
        jwtToken: state.jwtToken,
        user: state.user,
        strategies: state.strategies
      })
    }
  )
);
```
